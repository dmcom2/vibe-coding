핵심 기술 ① — Canvas API

5가지 핵심 메서드로 모든 도형 그리기
const canvas = document.getElementById('board'); const ctx = canvas.getContext('2d'); // ── 자유 드로잉 ────────────────────────── ctx.lineWidth = 3; ctx.strokeStyle = '#1e293b'; ctx.lineCap = 'round'; // 선 끝을 둥글게 ctx.beginPath(); ctx.moveTo(x1, y1); // 시작점 ctx.lineTo(x2, y2); // 끝점 ctx.stroke(); // 선 렌더링 // ── 원 그리기 ──────────────────────────── ctx.beginPath(); ctx.arc(cx, cy, radius, 0, Math.PI * 2); // 0 ~ 2π = 완전한 원 ctx.stroke(); // ── 직사각형 ───────────────────────────── ctx.strokeRect(x, y, width, height); // 외곽선만 ctx.fillRect(x, y, width, height); // 채우기 // ── 전체 지우기 ────────────────────────── ctx.clearRect(0, 0, canvas.width, canvas.height);
💡 Canvas는 beginPath → 형태 정의 → stroke/fill 3단계. beginPath 없이 연속으로 그리면 이전 경로와 연결됩니다. 항상 새 도형 전에 beginPath 호출.

핵심 기술 ② — 듀얼 캔버스 패턴

영구 캔버스 + 미리보기 캔버스를 겹쳐 성능 최적화
/* CSS: 두 캔버스를 같은 위치에 겹치기 */ /* #board { position: absolute; } 영구 저장용 */ /* #overlay { position: absolute; 미리보기용 */ /* pointer-events: none; } 클릭 이벤트 통과! */ const board = document.getElementById('board'); const overlay = document.getElementById('overlay'); const bCtx = board.getContext('2d'); const oCtx = overlay.getContext('2d'); // mousedown: 시작점 기록 board.addEventListener('mousedown', e => { startX = e.offsetX; startY = e.offsetY; isDrawing = true; }); // mousemove: overlay에만 미리보기 (깜빡임 없이) board.addEventListener('mousemove', e => { if (!isDrawing) return; oCtx.clearRect(0, 0, overlay.width, overlay.height); drawShape(oCtx, startX, startY, e.offsetX, e.offsetY); }); // mouseup: board에 확정 그리기 + overlay 초기화 board.addEventListener('mouseup', e => { drawShape(bCtx, startX, startY, e.offsetX, e.offsetY); oCtx.clearRect(0, 0, overlay.width, overlay.height); saveHistory(); // Undo용 스냅샷 저장 isDrawing = false; });

핵심 기술 ③ — Undo/Redo 히스토리

toDataURL()로 캔버스 스냅샷을 배열에 저장
let history = []; // 스냅샷 배열 (base64 문자열) let historyIdx = -1; // 현재 위치 // 스냅샷 저장 (그리기 완료 후 호출) function saveHistory() { // Redo 가능 기록 제거 (새 그리기가 미래를 덮어씀) history = history.slice(0, historyIdx + 1); history.push(board.toDataURL()); // PNG base64로 변환 historyIdx = history.length - 1; } // Undo — 이전 스냅샷으로 복원 function undo() { if (historyIdx <= 0) return; historyIdx--; restoreSnapshot(history[historyIdx]); } // Redo — 다음 스냅샷으로 복원 function redo() { if (historyIdx >= history.length - 1) return; historyIdx++; restoreSnapshot(history[historyIdx]); } function restoreSnapshot(dataURL) { const img = new Image(); img.src = dataURL; img.onload = () => { bCtx.clearRect(0, 0, board.width, board.height); bCtx.drawImage(img, 0, 0); }; }
💡 toDataURL()은 캔버스 전체를 base64 PNG 문자열로 변환. 고해상도 캔버스는 문자열이 수 MB가 될 수 있으므로 히스토리 최대 개수(예: 50개) 제한 권장.

화살표 & 스티키 노트 드래그

Math.atan2로 방향 계산, mousedown/move/up으로 드래그
1

화살표 방향 계산

angle = Math.atan2(y2-y1, x2-x1) — 두 점 사이의 각도(라디안). 화살촉은 angle ± Math.PI/6 방향으로 짧은 선 두 개를 그립니다.

2

화살촉 그리기

끝점(x2, y2)에서 Math.cos(angle ± π/6) × 15, Math.sin(angle ± π/6) × 15를 빼면 화살촉 꼭짓점 위치 계산.

3

스티키 노트 드래그

mousedown → offsetX = e.clientX - el.offsetLeft. mousemove → el.style.left = e.clientX - offsetX + 'px'. mouseup → 이벤트 해제.

4

PNG 내보내기

board.toBlob(blob => { a.href = URL.createObjectURL(blob); a.click() }) — blob URL로 직접 다운로드.

도구 선택 & 그리기 분기 패턴

현재 선택된 도구에 따라 다른 그리기 함수 호출
let currentTool = 'pen'; function drawShape(ctx, x1, y1, x2, y2) { ctx.beginPath(); switch(currentTool) { case 'pen': ctx.moveTo(x1, y1); ctx.lineTo(x2, y2); break; case 'rect': ctx.strokeRect(x1,y1,x2-x1,y2-y1); return; case 'circle': const r = Math.hypot(x2-x1,y2-y1); ctx.arc(x1,y1,r,0,Math.PI*2); break; case 'arrow': drawArrow(ctx, x1, y1, x2, y2); return; } ctx.stroke(); }

도구 선택 UI

도구 버튼 클릭 → currentTool = btn.dataset.tool. 활성 버튼에 CSS 클래스 추가로 시각적 강조.

지우개 구현

ctx.globalCompositeOperation = 'destination-out'으로 선 색상 대신 투명도를 사용해 기존 픽셀 제거.
