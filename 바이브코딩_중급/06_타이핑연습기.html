WPM 계산 공식

Words Per Minute — 분당 단어 수를 정밀하게 계산하는 방법

// WPM 계산 공식 // WPM = (입력 글자 수 / 5) / 경과 분 // 5글자 = 1단어로 표준화 let startTime = null; let interval = null; function startTyping() { // performance.now(): 밀리초 단위 고정밀 타이머 startTime = performance.now(); // 100ms마다 WPM 업데이트 interval = setInterval(updateStats, 100); } function updateStats() { const elapsed = performance.now() - startTime; const minutes = elapsed / 60000; // ms → 분 const charCount = typed.length; // 입력한 글자 수 const wpm = Math.round( (charCount / 5) / minutes ); document.getElementById('wpm').textContent = wpm; document.getElementById('time').textContent = Math.floor(elapsed / 1000) + '초'; }
Date.now() vs performance.now()

• Date.now() — 유닉스 타임스탬프 (ms)
시스템 시간이 바뀌면 오차 발생

• performance.now() — 페이지 로드 이후 경과 ms
시스템 영향 없음, 소수점 정밀도

타이핑 측정처럼 정밀도가 필요할 때는 performance.now() 사용!

WPM 예시 계산

30초 동안 150글자 입력 시:
• 분 수 = 30/60 = 0.5분
• 단어 수 = 150/5 = 30단어
• WPM = 30 / 0.5 = 60 WPM

실시간 색상 피드백

입력한 글자와 제시문을 비교해서 글자마다 정오 색상 표시

function renderText(typed) { const target = currentText; let html = ''; for (let i = 0; i < target.length; i++) { const ch = target[i]; if (i < typed.length) { // 입력된 글자: 정오 판별 if (typed[i] === ch) { html += `<span class="correct">${ch}</span>`; } else { html += `<span class="wrong">${ch}</span>`; wrongCount++; // 오타 누적 } } else if (i === typed.length) { // 현재 커서 위치 html += `<span class="cursor">${ch}</span>`; } else { // 아직 입력 안 된 글자 html += `<span class="pending">${ch}</span>`; } } textDisplay.innerHTML = html; }

프롬프트 해설 — 왜 잘 작동했을까?

앞 슬라이드 프롬프트를 분석해 "AI에게 잘 말하는 법"을 배웁니다

✅ POINT 1

섹션으로 나눠 명세하기

프롬프트를 "화면 구성 / 기능 / 디자인"으로 구분해 작성했습니다. AI가 전체 구조를 한눈에 파악하므로 빠뜨리거나 뒤섞는 실수가 없어집니다.

✅ POINT 2

시각 규칙을 색상 코드로 명시

"맞음: 초록색, 틀림: 빨간색 배경, 미입력: 회색, 커서: 파란 언더라인"처럼 색상을 직접 지정했습니다. AI가 디자인을 임의로 결정하지 않고 정확히 구현합니다.

✅ POINT 3

트리거→결과 쌍으로 기술

"첫 글자 입력 시→타이머 시작", "목표 텍스트 끝까지 입력 시→결과 화면 표시"처럼 조건과 동작을 함께 씁니다. AI가 언제 무엇을 해야 할지 혼동하지 않습니다.

✅ POINT 4

기술 조건을 마지막에 한 번에

"파일 3개로 나눠서, 한국어 주석 달아줘"처럼 파일 구조와 코드 품질 요구사항을 마지막에 묶어 명시하면 AI가 처음부터 올바른 구조로 코드를 작성합니다.

정확도 계산

전체 입력 횟수 대비 정타 비율을 누적으로 추적

let totalKeystrokes = 0; // 전체 입력 횟수 (백스페이스 제외) let wrongKeystrokes = 0; // 틀린 입력 횟수 textarea.addEventListener('input', e => { const typed = e.target.value; // 방금 입력한 글자와 제시문 비교 const lastIdx = typed.length - 1; const lastTyped = typed[lastIdx]; const targetChar = currentText[lastIdx]; if (lastTyped !== undefined) { totalKeystrokes++; if (lastTyped !== targetChar) wrongKeystrokes++; } renderText(typed); updateAccuracy(); // 완료 감지 if (typed === currentText) finishTyping(); }); function updateAccuracy() { if (totalKeystrokes === 0) return; const acc = Math.round( (1 - wrongKeystrokes / totalKeystrokes) * 100 ); document.getElementById('accuracy').textContent = acc + '%'; }
